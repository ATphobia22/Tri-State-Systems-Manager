/*
 * PTDT-v33: Archimedes ≥ WebGPU Compute Coupler
 * Evaluates high-performance Eulerian-Lagrangian fluid displacement and 
 * buoyant pressure forces directly on the GPU for sub-grid hydrodynamic interaction.
 */

struct FluidCell {
    velocity: vec3<f32>,
    pressure: f32,
};

struct FloatingBody {
    position: vec3<f32>,
    volume: f32,
    velocity: vec3<f32>,
    buoyancy_factor: f32,
};

@group(0) @binding(0) var<storage, read> fluid_grid: array<FluidCell>;
@group(0) @binding(1) var<storage, read_write> bodies: array<FloatingBody>;

struct SimParams {
    grid_width: u32,
    grid_height: u32,
    delta_time: f32,
    water_density: f32,
};
@group(0) @binding(2) var<uniform> params: SimParams;

@compute @workgroup_size(64, 1, 1)
fn main(@builtin(global_invocation_id) global_id: vec3<u32>) {
    let index = global_id.x;
    if (index >= arrayLength(&bodies)) {
        return;
    }

    var body = bodies[index];

    // Compute Archimedes' Principle: Upward buoyant force proportional to displaced fluid mass
    // F_buoyant = rho * V * g (applied along Y axis)
    let upward_acceleration = params.water_density * body.volume * 9.81 * body.buoyancy_factor;
    
    // Apply hydrodynamic drag from local fluid velocity field
    // Simplified 1D/3D index lookup for fluid cell interaction
    body.velocity.y += (upward_acceleration / (body.volume * 1000.0)) * params.delta_time;
    
    // Damping
    body.velocity *= 0.98;
    body.position += body.velocity * params.delta_time;

    bodies[index] = body;
}
